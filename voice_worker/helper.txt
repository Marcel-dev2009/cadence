This service takes the confirmation text generated by your application (for example: "Setting your theme to dark mode" or "Creating event: code review") and transforms it into a binary audio stream that Flutter can playback to the user.



import base64
from fastapi import FastAPI, UploadFile, File, HTTPException
from services import stt_service, tts_service

app = FastAPI()

@app.post("/api/voice/process")
async def process_voice_file(file: UploadFile = File(...)):
    if not file:
        raise HTTPException(status_code=400, detail="No audio file provided.")
        
    try:
        audio_bytes = await file.read()
        
        # 1. Sound -> Text -> Structural AI JSON
        structured_payload = stt_service.transcribe_and_route(audio_bytes)
        
        # 2. Build a customized confirmation reply for the user based on intent
        intent = structured_payload.get("intent", "unknown")
        if intent == "create_event":
            reply_text = f"Got it, scheduling {structured_payload.get('event_name')}."
        elif intent == "change_theme":
            reply_text = f"Switching your screen to {structured_payload.get('value')} mode."
        elif intent == "open_settings":
            reply_text = "Opening your account configurations."
        elif intent == "todo_list":
            reply_text = f"Creating a new checklist for {structured_payload.get('goal_name')}."
        else:
            reply_text = "I'm not sure how to handle that request yet."
            
        # 3. Generate the response audio file bytes
        reply_audio_bytes = tts_service.synthesize_speech(reply_text)
        
        # 4. Pack both into the final response payload safely
        structured_payload["voice_reply"] = base64.b64encode(reply_audio_bytes).decode('utf-8')
        structured_payload["spoken_text"] = reply_text
        
        return structured_payload
        
    except Exception as e:
        raise HTTPException(status_code=500, detail=str(e))
    finally:
        await file.close()
